// !Truthy and Falsy values
// If a value is evaluated(coerced) to true in a boolean context that value is a truthy value 
// If a value is evaluated(coerced) to false in a boolean context that value is a falsy value 
// ! There are 7 falsy values in js
// 0
// 0n
// ''
// false
// undefined
// null
// NaN
// ! other than these 7 values , everything is truthy
// To check if a value is falsy or truthy we can use Boolean()
console.log(Boolean(0)); // false
console.log(Boolean('Atul')); // true
console.log(Boolean([])); // true
console.log(Boolean(null)); // false
console.log(Boolean(undefined)); //false
console.log(Boolean(NaN)); //false
console.log(Boolean(30)); //true
